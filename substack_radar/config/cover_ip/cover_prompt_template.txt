# TEMPLATE (純文字，直接貼進 image_brain)

COVER_PROMPT_TEMPLATE = """\
{character_block}
Scene: {scene}
Composition: single subject, placed on the {anchor} third of the frame, \
{gaze} toward the empty title zone; leave the opposite 55-60% as clean \
negative space for a Chinese headline overlay (do not render any text). \
Character occupies 32-42% of frame height, feet/base near lower third. \
Eye-line and any magnifier/gesture point INTO the title zone, guiding the reader.
{style_tail}
Aspect ratio 16:9, 1456x816. No text, no watermark, no logo, no border, \
no neon, no gradient, no 3D glow, no anime face. Paper-cream #F2EEE5 ground only.
"""

# {anchor}: "left" | "right"   — which side the character hugs (title goes opposite)
# {gaze}:   "looking right" | "looking left" — must point toward title zone

# ─────────────────────────────────────────────────────────────────────────────
# FIXED CHARACTER BLOCKS  (= image_brain.CHARACTERS[*]["look"], 已鎖 v1，勿改)
# ─────────────────────────────────────────────────────────────────────────────

A chunky rounded desk-robot analyst made of soft matte clay, stone-grey #8A8378 \
body, a small spinning radar dish antenna on its head emitting a tiny "ping!" \
spark, one big glossy single lens-eye that sparkles, stubby articulated arms, a \
sienna-red #C84A32 knitted scarf. Squash-and-stretch rubbery posing, exaggerated \
and lively."""

OWL_LOOK = """\
A plump rounded owl made of soft matte clay, warm stone-grey #8A8378 feathers with \

sienna-red #C84A32 bow-tie scarf, stubby wings. Feathers puffed up, very \
expressive, theatrical squash-and-stretch posing."""
